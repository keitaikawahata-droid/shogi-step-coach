import subprocess import re import time from typing import Optional, Tuple, Dict, Any class EngineConnector: """ UCI/USIプロトコルに対応した外部思考エンジンを制御するクラス。 チェス(Stockfish等)や将棋(やねうら王等)のバイナリを呼び出して評価値を取得します。 """ def __init__(self, engine_path: str, protocol: str = "uci"): self.engine_path = engine_path self.protocol = protocol.lower() # "uci" または "usi" self.process: Optional[subprocess.Popen] = None self._start_engine() def _start_engine(self) -> None: # エンジンプロセスを標準入出力パイプで起動 self.process = subprocess.Popen( [self.engine_path], stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True, bufsize=1 ) # 初期化コマンドの送信 init_cmd = "uci" if self.protocol == "uci" else "usi" ok_response = "uciok" if self.protocol == "uci" else "usiok" self.send_command(init_cmd) self._wait_for_response(ok_response) # 準備完了の確認 self.send_command("isready") self._wait_for_response("readyok") def send_command(self, command: str) -> None: if self.process and self.process.stdin: self.process.stdin.write(f"{command}\n") self.process.stdin.flush() def _wait_for_response(self, target: str) -> None: if not self.process or not self.process.stdout: return while True: line = self.process.stdout.readline().strip() if line == target: break def evaluate_position(self, moves_str: str = "", search_time_ms: int = 1000) -> Dict[str, Any]: """ 指定した着手系列（例: 'e2e4 e7e5' や '7g7f 3c3d'）の局面を探索し、 評価値と最善手を返します。 """ if not self.process or not self.process.stdout: raise RuntimeError("エンジンが起動していません。") # 局面のセット if moves_str.strip(): self.send_command(f"position startpos moves {moves_str.strip()}") else: self.send_command("position startpos") # 思考開始コマンドの送信（指定ミリ秒探索） self.send_command(f"go movetime {search_time_ms}") latest_score_cp: Optional[int] = None latest_mate: Optional[int] = None bestmove: Optional[str] = None # エンジンからの出力を1行ずつ解析 while True: line = self.process.stdout.readline().strip() if not line: continue # 評価値情報行（info depth ... score cp 120 ...）のパース if line.startswith("info") and "score" in line: cp_match = re.search(r"score\s+cp\s+(-?\d+)", line) mate_match = re.search(r"score\s+mate\s+(-?\d+)", line) if cp_match: latest_score_cp = int(cp_match.group(1)) latest_mate = None elif mate_match: latest_mate = int(mate_match.group(1)) latest_score_cp = None # 思考完了行（bestmove e7e5 ...） if line.startswith("bestmove"): parts = line.split() if len(parts) >= 2: bestmove = parts[1] break return { "score_cp": latest_score_cp, # センティポーン単位（100 = ポーン/歩1枚分の有利） "mate_in": latest_mate, # 詰みまでの手数（詰み局面でない場合はNone） "bestmove": bestmove # 最善手 } def quit(self) -> None: if self.process: self.send_command("quit") self.process.terminate() self.process.wait() if __name__ == "__main__": # 使用例： # 1. 外部エンジンの実行ファイルパスを指定（環境に合わせて変更してください） # 例: "./stockfish/stockfish-ubuntu-x86-64-avx2" や "C:/engines/stockfish.exe" ENGINE_PATH = "./stockfish" try: # チェスエンジンの場合: protocol="uci" # 将棋エンジンの場合: protocol="usi" engine = EngineConnector(engine_path=ENGINE_PATH, protocol="uci") print("エンジンを初期化しました。初期局面から1手目 'e2e4' 後の評価値を計算します...") # 初期局面から 'e2e4' が指された局面の評価（探索時間1.5秒） result = engine.evaluate_position(moves_str="e2e4", search_time_ms=1500) print("\n--- 解析結果 ---") if result["score_cp"] is not None: print(f"評価値 (先手視点): {result['score_cp']} cp") elif result["mate_in"] is not None: print(f"詰み: {result['mate_in']} 手詰み") print(f"エンジンの推奨する最善手: {result['bestmove']}") engine.quit() except FileNotFoundError: print(f"指定されたパスにエンジンが見つかりません: {ENGINE_PATH}") print("エンジンのバイナリをダウンロードし、実行パスを正しく設定してください。")
